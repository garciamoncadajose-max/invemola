// Versión ultra-simplificada nativa para desarrollo local sin internet
var XLSX = {
    utils: {
        json_to_sheet: function(data) { return data; },
        book_new: function() { return { Sheets: {}, SheetNames: [] }; },
        book_append_sheet: function(wb, sh, name) { wb.Sheets[name] = sh; wb.SheetNames.push(name); }
    },
    writeFile: function(wb, filename) {
        var csv = "";
        var name = wb.SheetNames[0];
        var data = wb.Sheets[name];
        if(!data || data.length === 0) return;
        var headers = Object.keys(data[0]);
        csv += headers.join(",") + "\r\n";
        data.forEach(function(row) {
            var line = headers.map(function(h) { 
                var val = row[h] === undefined ? "" : row[h];
                return '"' + String(val).replace(/"/g, '""') + '"';
            });
            csv += line.join(",") + "\r\n";
        });
        var blob = new Blob(["\ufeffsep=,\r\n" + csv], { type: 'text/csv;charset=utf-8;' });
        var link = document.createElement("a");
        link.href = URL.createObjectURL(blob);
        link.setAttribute("download", filename.replace(".xlsx", ".csv"));
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }
};
